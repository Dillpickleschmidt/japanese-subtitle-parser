<project_specification>
  <project_name>Japanese Grammar Pattern Library - JLPT Detection System</project_name>

  <overview>
    A Rust library for detecting JLPT (Japanese Language Proficiency Test) grammar patterns in Japanese text.
    The library tokenizes Japanese sentences using Kagome morphological analyzer, then identifies grammar
    patterns ranging from N5 (beginner) to N1 (advanced). The goal is to implement all patterns
    with proper test coverage following the conventions in REFERENCE_EXAMPLES.md.
  </overview>

  <technology_stack>
    <language>Rust</language>
    <tokenizer>Kagome (Go-based morphological analyzer, runs on port 6061)</tokenizer>
    <database>SQLite (for application integration)</database>
    <testing>cargo test</testing>
    <dependencies>
      - serde: JSON serialization
      - kagome_client: HTTP client for Kagome tokenizer
    </dependencies>
  </technology_stack>

  <prerequisites>
    - Rust toolchain installed (cargo, rustc)
    - Kagome server running on localhost:6061
    - Run `cargo build` to verify compilation
    - Run `cargo test` to run test suite
  </prerequisites>

  <architecture>
    <directory_structure>
      grammar-lib/
      ├── src/
      │   ├── lib.rs                    # Main library entry point and analyze() function
      │   ├── types.rs                  # Core types: KagomeToken, AnalysisResult
      │   ├── patterns.rs               # Pattern declarations via declare_patterns! macro
      │   ├── pattern_matcher.rs        # Pattern matching engine
      │   ├── pattern_registry.rs       # JLPT level registry
      │   ├── matchers/
      │   │   ├── mod.rs                # Common matchers and Matcher trait
      │   │   ├── n5.rs                 # N5 level matcher functions
      │   │   ├── n4.rs                 # N4 level matcher functions
      │   │   ├── n3.rs                 # N3 level matcher functions
      │   │   ├── n2.rs                 # N2 level matcher functions
      │   │   └── n1.rs                 # N1 level matcher functions
      │   └── tests/
      │       ├── mod.rs                # Test helpers: tokenize_sentence, detect_patterns, assert_*
      │       ├── n5_patterns.rs        # N5 pattern tests
      │       ├── n4_patterns.rs        # N4 pattern tests
      │       ├── n3_patterns.rs        # N3 pattern tests
      │       ├── n2_patterns.rs        # N2 pattern tests
      │       └── n1_patterns.rs        # N1 pattern tests
      └── Cargo.toml
    </directory_structure>

    <pattern_declaration>
      Patterns are declared in patterns.rs using the declare_patterns! macro:
      ```rust
      declare_patterns! {
          PatternName {
              name: "pattern_name",
              matcher_fn: matcher_function,
              priority: 6,
              category: PatternCategory::Construction,
              jlpt: "n3",
          },
      }
      ```
    </pattern_declaration>

    <token_matcher_types>
      - TokenMatcher::Verb { conjugation_form, base_form } - Match verbs
      - TokenMatcher::Surface("text") - Exact surface match
      - TokenMatcher::Custom(Arc::new(MatcherStruct)) - Custom logic
      - TokenMatcher::Optional(Box::new(...)) - Optional token
      - TokenMatcher::Wildcard { min, max, stop_conditions } - Skip tokens
      - TokenMatcher::Any - Match any single token
    </token_matcher_types>

    <test_helpers>
      Key test helper functions in tests/mod.rs:
      - tokenize_sentence(text) - Tokenize Japanese text via Kagome
      - detect_patterns(&tokens) - Detect grammar patterns in tokens
      - has_pattern(&matches, name) - Check if pattern exists
      - assert_has_pattern(&matches, name) - Assert pattern detection
      - assert_pattern_range(&matches, name, start, end) - Assert character positions
      - print_debug(sentence, &tokens, &patterns) - Debug output (remove before committing)
    </test_helpers>
  </architecture>

  <jlpt_levels>
    <n5>Basic grammar patterns. Foundation for all other levels. (~50 patterns)</n5>
    <n4>Elementary grammar. Te-form extensions, conditionals, causative. (~63 patterns)</n4>
    <n3>Intermediate grammar. Complex constructions, adverbs. (~62 patterns)</n3>
    <n2>Upper intermediate. Formal/literary patterns. (~73 patterns)</n2>
    <n1>Advanced grammar. Classical/literary forms. (~73 patterns)</n1>
  </jlpt_levels>

  <data_sources>
    <grammar_points_data>
      File: grammar_points_data.json
      Format: { "pattern_name": { structure: { standard: [...], polite: [...] }, register, about } }

      - structure.standard: Array of standard form structures
      - structure.polite: Array of polite form structures
      - about: HTML explanation with examples (note: `<target-japanese>` tags show minimal ranges,
        but patterns should include the full grammatical construction - see REFERENCE_EXAMPLES.md)

      Pattern ranges should include the **full grammatical construction** (e.g., verb + suffix, noun + particle).
      See prompts/REFERENCE_EXAMPLES.md for examples of correct ranges.
      Access via: `jq '.["pattern_name"]' grammar_points_data.json`
    </grammar_points_data>

    <grammar_points_csv>
      Files: grammar_points.csv, grammar_points_jlpt.csv
      Maps patterns to JLPT levels and completion status.
    </grammar_points_csv>
  </data_sources>

  <success_criteria>
    A pattern is complete when:
    - Test(s) exist using assert_pattern_range() (not just assert_has_pattern())
    - No print_debug() calls in committed tests
    - Test sentences are realistic (subtitle-quality Japanese)
    - Character positions are correct (0-indexed)
  </success_criteria>
